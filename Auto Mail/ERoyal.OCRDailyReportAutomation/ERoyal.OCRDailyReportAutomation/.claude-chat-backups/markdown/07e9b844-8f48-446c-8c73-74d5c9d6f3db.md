        // Append DataTable rows to worksheet (with header on first call)

        private static void AppendDataTable(IXLWorksheet ws, DataTable table, ref int currentRow)
        {
            if (table == null || table.Rows.Count == 0) return;

            // Write header only on first call (row 1)
            if (currentRow == 1)
            {
                for (int col = 0; col < table.Columns.Count; col++)
                {
                    var cell = ws.Cell(currentRow, col + 1);
                    cell.Value = table.Columns[col].ColumnName;
                    cell.Style.Font.Bold = true;
                    cell.Style.Fill.BackgroundColor = XLColor.DarkBlue;
                    cell.Style.Font.FontColor       = XLColor.White;
                    cell.Style.Alignment.Horizontal = XLAlignmentHorizontalValues.Center;
                }
                currentRow++;
            }

            foreach (DataRow row in table.Rows)
            {
                for (int col = 0; col < table.Columns.Count; col++)
                {
                    ws.Cell(currentRow, col + 1).Value = row[col]?.ToString();
                }

                // Alternate row color
                if (currentRow % 2 == 0)
                {
                    ws.Range(currentRow, 1, currentRow, table.Columns.Count)
                      .Style.Fill.BackgroundColor = XLColor.LightGray;
                }

                currentRow++;
            }
        }

        // ─────────────────────────────────────────────────────────────────────
        // Parse dd-MM-yyyy → yyyy-MM-dd for SP date params
        // ─────────────────────────────────────────────────────────────────────
        private static string ParseDateToYMD(string dateStr)
        {
            if (string.IsNullOrWhiteSpace(dateStr)) return "";
            try
            {
                return DateTime.ParseExact(dateStr.Trim(), "dd-MM-yyyy", null)
                               .ToString("yyyy-MM-dd");
            }
            catch
            {
                Logger.Warn("Could not parse date: {0}", dateStr);
                return dateStr;
            }
        }

        private static async Task SendEmailWithAttachment(string filePath)
        {
            try
            {
                Logger.Info("Sending email...");

                using var mail = new MailMessage
                {
                    From       = new MailAddress(_smtpSettings.FromEmail, _smtpSettings.Mail_header),
                    Subject    = _smtpSettings.Subject.Replace("{Date}", DateTime.Now.ToString("dd-MM-yyyy")),
                    Body       = _smtpSettings.Body,
                    IsBodyHtml = true
                };

                foreach (var to in (_smtpSettings.ToEmails ?? Array.Empty<string>()))
                    mail.To.Add(to.Trim());

                foreach (var cc in (_smtpSettings.CCEmails ?? Array.Empty<string>()))
                    mail.CC.Add(cc.Trim());

                using (var fs = new FileStream(filePath, FileMode.Open, FileAccess.Read, FileShare.Read))
                {
                    var attachment = new Attachment(fs, Path.GetFileName(filePath));
                    mail.Attachments.Add(attachment);

                    using (var smtp = new SmtpClient(_smtpSettings.Host, _smtpSettings.SmtpPort))
                    {
                        smtp.Credentials = new NetworkCredential(_smtpSettings.username, _smtpSettings.password);
                        smtp.EnableSsl   = _smtpSettings.enableSsl;
                        await smtp.SendMailAsync(mail);
                    }
                }

                Logger.Info("Email sent: {0}", Path.GetFileName(filePath));

                if (File.Exists(filePath))
                {
                    File.Delete(filePath);
                    Logger.Info("Deleted report: {0}", filePath);
                }
            }
            catch (Exception ex)
            {
                Logger.Error(ex, "Email send failed");
            }
        }

        private static void AppendDataTable(IXLWorksheet ws, DataTable table, ref int currentRow)
        {
            if (table == null || table.Rows.Count == 0) return;

            if (currentRow == 1)
            {
                for (int col = 0; col < table.Columns.Count; col++)
                {
                    ws.Cell(currentRow, col + 1).Value = table.Columns[col].ColumnName;
                    ws.Cell(currentRow, col + 1).Style.Font.Bold = true;
                }
                currentRow++;
            }

            foreach (DataRow row in table.Rows)
            {
                for (int col = 0; col < table.Columns.Count; col++)
                {
                    ws.Cell(currentRow, col + 1).Value = row[col]?.ToString();
                }
                currentRow++;
            }
