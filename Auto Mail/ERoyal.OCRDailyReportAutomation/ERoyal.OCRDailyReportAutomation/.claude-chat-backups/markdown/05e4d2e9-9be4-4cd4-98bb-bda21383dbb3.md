        private static void AppendDataTable(IXLWorksheet ws, DataTable table, ref int currentRow)
        {
            if (table == null || table.Rows.Count == 0) return;

            if (currentRow == 1)
            {
                for (int col = 0; col < table.Columns.Count; col++)
                {
                    ws.Cell(currentRow, col + 1).Value = table.Columns[col].ColumnName;
                    ws.Cell(currentRow, col + 1).Style.Font.Bold = true;
                }
                currentRow++;
            }

            foreach (DataRow row in table.Rows)
            {
                for (int col = 0; col < table.Columns.Count; col++)
                {
                    ws.Cell(currentRow, col + 1).Value = row[col]?.ToString();
                }
                currentRow++;
            }
        }
